# /etc/nginx/conf.d/probiz.conf - Enhanced Security Version
server {
    listen 80;
    server_name 144.24.7.55 _;
    
    # Redirect HTTP to HTTPS
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name 144.24.7.55 _;
    
    # SSL Configuration (after getting certificate)
    ssl_certificate /etc/letsencrypt/live/144.24.7.55/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/144.24.7.55/privkey.pem;
    ssl_session_cache shared:SSL:1m;
    ssl_session_timeout 10m;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    
    # Security Headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval'; style-src 'self' 'unsafe-inline'; img-src 'self' data:; font-src 'self'; connect-src 'self'; media-src 'self';" always;
    
    # Hide nginx version
    server_tokens off;
    
    # Rate limiting
    limit_req_zone $binary_remote_addr zone=main:10m rate=10r/m;
    limit_req_zone $binary_remote_addr zone=api:10m rate=30r/m;
    
    # Set the document root
    root /home/opc/ProBiz/websites/main/public;
    index index.html index.htm;
    
    # Rate limit general requests
    limit_req zone=main burst=20 nodelay;
    
    # Disable autoindex by default (security)
    autoindex off;
    
    # Block access to hidden files
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }
    
    # Block access to backup files
    location ~* \.(bak|backup|old|orig|tmp)$ {
        deny all;
        access_log off;
        log_not_found off;
    }
    
    # Main location block
    location / {
        try_files $uri $uri/ =404;
    }
    
    # Games directory with rate limiting
    location /games/ {
        alias /home/opc/ProBiz/websites/games/;
        try_files $uri $uri/ =404;
        limit_req zone=main burst=10 nodelay;
    }
    
    # Projects with authentication (optional)
    location /projects/ {
        alias /home/opc/ProBiz/websites/;
        try_files $uri $uri/ =404;
        
        # Uncomment to add basic auth
        # auth_basic "Restricted Area";
        # auth_basic_user_file /etc/nginx/.htpasswd;
    }
    
    # Admin areas with authentication
    location /docs/ {
        alias /home/opc/ProBiz/docs/;
        try_files $uri $uri/ =404;
        
        # Add basic authentication
        auth_basic "Documentation Access";
        auth_basic_user_file /etc/nginx/.htpasswd;
    }
    
    location /nodejs/ {
        alias /home/opc/ProBiz/nodejs/;
        try_files $uri $uri/ =404;
        
        # Add basic authentication
        auth_basic "Development Area";
        auth_basic_user_file /etc/nginx/.htpasswd;
    }
    
    # Utility directory
    location /utility/ {
        alias /home/opc/ProBiz/websites/utility/;
        try_files $uri $uri/ =404;
    }
    
    # EJ-EV project locations (same as before but with rate limiting)
    location /EJ-EV/ {
        alias /home/opc/ProBiz/websites/EJ-EV/;
        try_files $uri $uri/ =404;
        limit_req zone=main burst=10 nodelay;
    }
    
    # API with stricter rate limiting
    location /api/ {
        alias /home/opc/ProBiz/nodejs/;
        limit_req zone=api burst=50 nodelay;
    }
    
    # Static assets with caching
    location /shared/ {
        alias /home/opc/ProBiz/shared/;
        expires 1y;
        add_header Cache-Control "public, immutable";
    }
    
    # Gzip compression
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/javascript application/json image/svg+xml;
    
    # Logs
    error_log /var/log/nginx/probiz_error.log;
    access_log /var/log/nginx/probiz_access.log;
}
